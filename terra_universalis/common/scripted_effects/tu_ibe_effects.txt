TU_IBE_monthly = {
	add_to_variable = { TU_IBE_threat = 2 }
	set_temp_variable = { tu_ibe_lh = TU_IBE_lighthouses }
	multiply_temp_variable = { tu_ibe_lh = -0.5 }
	add_to_variable = { TU_IBE_threat = tu_ibe_lh }
	clamp_variable = { var = TU_IBE_threat min = 0 max = 100 }
}
