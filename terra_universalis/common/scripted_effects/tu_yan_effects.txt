# Called from common/on_actions/tu_yan_on_actions.txt (on_startup).
# Yan: six infantry divisions in the capital state (64) and two each in the
# adjacent Yan states 71 and 61 (ten in total).
TU_YAN_startup = {
	capital_scope = {
		create_unit = {
			division = "division_template = \"TU_YAN_infantry\" start_experience_factor = 0.2 start_equipment_factor = 0.7"
			owner = YAN
			count = 6
		}
	}
	if = {
		limit = {
			71 = { is_owned_by = YAN }
		}
		71 = {
			create_unit = {
				division = "division_template = \"TU_YAN_infantry\" start_experience_factor = 0.2 start_equipment_factor = 0.7"
				owner = YAN
				count = 2
			}
		}
	}
	if = {
		limit = {
			61 = { is_owned_by = YAN }
		}
		61 = {
			create_unit = {
				division = "division_template = \"TU_YAN_infantry\" start_experience_factor = 0.2 start_equipment_factor = 0.7"
				owner = YAN
				count = 2
			}
		}
	}
}
