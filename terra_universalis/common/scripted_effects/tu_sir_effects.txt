TU_SIR_monthly = {
	if = {
		limit = { has_country_flag = TU_SIR_route_bellone }
		set_variable = { TU_SIR_bellone = 100 }
	}
	else_if = {
		limit = { has_country_flag = TU_SIR_absorbed_bellone }
		set_variable = { TU_SIR_bellone = 0 }
	}
	else = {
		add_to_variable = { TU_SIR_bellone = 1 }
		clamp_variable = { var = TU_SIR_bellone min = 0 max = 100 }
	}
	if = {
		limit = { has_country_flag = TU_SIR_route_gambino }
		set_variable = { TU_SIR_gambino = 100 }
	}
	else_if = {
		limit = { has_country_flag = TU_SIR_absorbed_gambino }
		set_variable = { TU_SIR_gambino = 0 }
	}
	else = {
		add_to_variable = { TU_SIR_gambino = 1 }
		clamp_variable = { var = TU_SIR_gambino min = 0 max = 100 }
	}
	if = {
		limit = { has_country_flag = TU_SIR_route_saluzzo }
		set_variable = { TU_SIR_saluzzo = 100 }
	}
	else_if = {
		limit = { has_country_flag = TU_SIR_absorbed_saluzzo }
		set_variable = { TU_SIR_saluzzo = 0 }
	}
	else = {
		add_to_variable = { TU_SIR_saluzzo = 1 }
		clamp_variable = { var = TU_SIR_saluzzo min = 0 max = 100 }
	}
	if = { limit = { has_country_flag = TU_SIR_courts } add_to_variable = { TU_SIR_law = 1 } }
	clamp_variable = { var = TU_SIR_law min = 0 max = 100 }
	TU_SIR_unrest_tick = yes
}

TU_SIR_unrest_tick = {
	if = {
		limit = {
			is_ai = no
			has_country_flag = TU_SIR_families
			NOT = { has_country_flag = TU_SIR_unrest_active }
			OR = {
				AND = { NOT = { has_country_flag = TU_SIR_route_bellone } NOT = { has_country_flag = TU_SIR_absorbed_bellone } }
				AND = { NOT = { has_country_flag = TU_SIR_route_gambino } NOT = { has_country_flag = TU_SIR_absorbed_gambino } }
				AND = { NOT = { has_country_flag = TU_SIR_route_saluzzo } NOT = { has_country_flag = TU_SIR_absorbed_saluzzo } }
			}
		}
		add_to_variable = { TU_SIR_unrest_timer = 1 }
		if = {
			limit = { check_variable = { TU_SIR_unrest_timer > 3 } }
			set_variable = { TU_SIR_unrest_timer = 0 }
			random_list = {
				10 = {
					modifier = { factor = 0 OR = { has_country_flag = TU_SIR_route_bellone has_country_flag = TU_SIR_absorbed_bellone } }
					modifier = { factor = 2 check_variable = { TU_SIR_bellone > 49 } }
					country_event = { id = tu_sir_ev.60 }
				}
				10 = {
					modifier = { factor = 0 OR = { has_country_flag = TU_SIR_route_gambino has_country_flag = TU_SIR_absorbed_gambino } }
					modifier = { factor = 2 check_variable = { TU_SIR_gambino > 49 } }
					country_event = { id = tu_sir_ev.61 }
				}
				10 = {
					modifier = { factor = 0 OR = { has_country_flag = TU_SIR_route_saluzzo has_country_flag = TU_SIR_absorbed_saluzzo } }
					modifier = { factor = 2 check_variable = { TU_SIR_saluzzo > 49 } }
					country_event = { id = tu_sir_ev.62 }
				}
			}
		}
	}
}

TU_SIR_unrest_won = {
	if = { limit = { has_country_flag = TU_SIR_unrest_by_bellone } TU_SIR_bellone_m10 = yes }
	if = { limit = { has_country_flag = TU_SIR_unrest_by_gambino } TU_SIR_gambino_m10 = yes }
	if = { limit = { has_country_flag = TU_SIR_unrest_by_saluzzo } TU_SIR_saluzzo_m10 = yes }
	army_experience = 10
	add_stability = 0.01
	TU_SIR_unrest_clear = yes
}

}
