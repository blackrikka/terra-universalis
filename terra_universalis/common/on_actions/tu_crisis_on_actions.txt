on_actions = {
	on_startup = {
		effect = {
			# Schedules the Chernobog incident once per save (flag prevents a second schedule).
			if = {
				limit = {
					country_exists = URS
					NOT = { has_global_flag = TU_chernobog_incident_scheduled }
				}
				set_global_flag = TU_chernobog_incident_scheduled
				# Countdown mission (spec 29): tu_crisis.1 fires when TU_URS_uprising_mission times out.
				URS = {
					activate_mission = TU_URS_uprising_mission
				}
			}
		}
	}
}
