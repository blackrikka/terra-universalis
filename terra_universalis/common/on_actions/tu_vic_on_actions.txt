on_actions = {
	on_startup = {
		effect = {
			# Once only (spec 40): on_startup can run again in multiplayer or on load
			if = {
				limit = { NOT = { has_global_flag = TU_VIC_startup_done } }
				set_global_flag = TU_VIC_startup_done
				if = {
					limit = {
						country_exists = VIC
					}
					VIC = {
						TU_VIC_startup = yes
					}
				}
			}
		}
	}
}
