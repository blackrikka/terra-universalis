TU_RHO_riic_category = {

	# Factory: offsite arms factory. Cap = TU_RHO_riic_cap (power plant level + 2).
	TU_RHO_upgrade_factory = {
		icon = GFX_decision_generic_factory
		allowed = { original_tag = RHO }
		available = {
			check_variable = { TU_RHO_riic_factory < TU_RHO_riic_cap }
			check_variable = { var = TU_lmd value = TU_RHO_riic_factory_cost compare = greater_than_or_equals }
		}
		cost = 25
		days_re_enable = 30
		ai_will_do = { factor = 0 }
		complete_effect = {
			subtract_from_variable = { TU_lmd = TU_RHO_riic_factory_cost }
			add_to_variable = { TU_RHO_riic_factory = 1 }
			add_to_variable = { TU_RHO_riic_factory_cost = 100 }
			add_offsite_building = { type = arms_factory level = 1 }
		}
	}

	# Trading post: offsite industrial complex (LMD income is in TU_RHO_update_income).
	TU_RHO_upgrade_trade = {

		}

	# Power plant: only raises the cap for the other facilities (max level 3).
	TU_RHO_upgrade_power = {

			check_variable = { TU_RHO_riic_power < 3 }

			add_to_variable = { TU_RHO_riic_power_cost = 100 }
			add_to_variable = { TU_RHO_riic_cap = 1 }
		}
	}

	# Dormitory: army organisation regain and morale via TU_RHO_riic_modifier.
	TU_RHO_upgrade_dorm = {
		icon = GFX_decision_generic_civil_support

		available = {
			check_variable = { TU_RHO_riic_dorm < TU_RHO_riic_cap }

		}
		cost = 25
		days_re_enable = 30
		ai_will_do = { factor = 0 }
		complete_effect = {
			subtract_from_variable = { TU_lmd = TU_RHO_riic_dorm_cost }
			add_to_variable = { TU_RHO_riic_dorm = 1 }
			add_to_variable = { TU_RHO_riic_dorm_cost = 100 }
			add_to_variable = { TU_RHO_riic_dorm_bonus = 0.03 }
			add_dynamic_modifier = { modifier = TU_RHO_riic_modifier }

	}

	# Reception room: political power gain via TU_RHO_riic_modifier.

			add_to_variable = { TU_RHO_riic_reception_cost = 100 }
			add_to_variable = { TU_RHO_riic_reception_bonus = 0.05 }

				limit = { has_dlc = "La Resistance" }
